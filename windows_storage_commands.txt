# Windows PowerShell Commands for Laravel Storage Setup

# Navigate to Laravel backend directory
cd backadm-dmc

# Create missing Laravel storage directories
New-Item -ItemType Directory -Path "storage\framework\cache\data" -Force
New-Item -ItemType Directory -Path "storage\framework\views" -Force
New-Item -ItemType Directory -Path "storage\framework\sessions" -Force
New-Item -ItemType Directory -Path "storage\framework\testing" -Force

# Create .gitkeep files to ensure folders are tracked in Git
New-Item -ItemType File -Path "storage\framework\cache\data\.gitkeep" -Force
New-Item -ItemType File -Path "storage\framework\views\.gitkeep" -Force
New-Item -ItemType File -Path "storage\framework\sessions\.gitkeep" -Force
New-Item -ItemType File -Path "storage\framework\testing\.gitkeep" -Force

# Return to root directory
cd ..
